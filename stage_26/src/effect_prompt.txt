    uint32_t now = millis();

        }
        break;

        uint16_t currentPos = (activeTime * SLAVE_STRIP_LEN) / SLAVE_WAVE_MS;
        if (currentPos >= SLAVE_STRIP_LEN) currentPos = SLAVE_STRIP_LEN - 1;

        CRGB* strip = stripPtrs[s];

        // RENDER STEP: Drive the anti-aliased trailing impulse front down the target strip
        // Keep bounds operations safely protected (e.g., currentPos - tailIndex >= 0)
      }
    }
    // ── Settle Glow & Cool-down ──
    else if (phase == NFXS_SETTLE) {

      uint8_t coolDownGlow = scale8(64, 255 - factor);

      if (coolDownGlow == 0) return;
